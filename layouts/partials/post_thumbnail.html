{{/* post_thumbnail.html — override of mainroad's default partial. Adds Hugo image processing for page-bundle posts (images stored alongside index.md as page resources). Static images (legacy posts in static/) fall through unchanged. Sizes: list → .Fill "300x200" — landscape 3:2 crop, uniform cards in the listing post → .Resize "900x" — wide, full image displayed block above content Per-post frontmatter (map form): thumbnail: src: "cover.jpg" visibility: ["list", "post"] alt: "Custom alt text" caption: "Credit: Generated by Google Gemini" Global default (hugo.toml): [Params.thumbnail] visibility = ["list"] */}} {{- if $thumbParam := .page.Params.thumbnail }} {{- $class := .class }} {{- $visibility := .page.Site.Params.thumbnail.visibility | default (slice "list" "post") }} {{- $alt := .page.Title }} {{- $caption := "" }} {{/* Support map-style thumbnail */}} {{- $thumbSrc := $thumbParam -}} {{- if in (printf "%T" $thumbParam) "map" }} {{- $visibility = default $visibility (default .page.Site.Params.thumbnail.visibility $thumbParam.visibility) }} {{- $alt = default .page.Title $thumbParam.alt }} {{- $caption = default "" $thumbParam.caption }} {{- $thumbSrc = $thumbParam.src }} {{- end }} {{- if in $visibility $class }} {{/* Resolve image: page-bundle resource → Hugo processing; static → passthrough */}} {{- $imgSrc := $thumbSrc | relURL }} {{- $resource := .page.Resources.GetMatch $thumbSrc }} {{- if $resource }} {{- if eq $class "list" }} {{- $resource = $resource.Fill "300x200 q85 Center" }} {{- else }} {{- $resource = $resource.Resize "900x q85" }} {{- end }} {{- $imgSrc = $resource.RelPermalink }} {{- end }} {{- end }} {{- end }}
